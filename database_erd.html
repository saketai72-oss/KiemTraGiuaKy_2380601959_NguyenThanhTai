-- =========================================================================
-- TOEIC EXAM DATABASE SCHEMA (MS SQL SERVER 2022)
-- =========================================================================

-- 1. BẢNG EXAMS (Đề thi)
CREATE TABLE Exams (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Title NVARCHAR(255) NOT NULL,
    DurationMinutes INT NOT NULL DEFAULT 75,
    TotalQuestions INT NOT NULL DEFAULT 100,
    IsPublished BIT NOT NULL DEFAULT 0,
    CreationTime DATETIME2 NOT NULL DEFAULT GETUTCDATE()
);

-- 2. BẢNG QUESTION GROUPS (Đoạn văn / Nhóm câu hỏi Part 5, 6, 7)
CREATE TABLE QuestionGroups (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    ExamId INT NOT NULL,
    PartNumber INT NOT NULL, -- 5, 6, 7
    DirectionText NVARCHAR(MAX) NULL,
    PassageContentHtml NVARCHAR(MAX) NULL,
    PassageImagesJson NVARCHAR(MAX) NULL,
    OrderIndex INT NOT NULL DEFAULT 0,
    CONSTRAINT FK_QuestionGroups_Exams FOREIGN KEY (ExamId) 
        REFERENCES Exams(Id) ON DELETE CASCADE
);

-- 3. BẢNG QUESTIONS (Câu hỏi 101 - 200)
CREATE TABLE Questions (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    ExamId INT NOT NULL,
    QuestionGroupId INT NOT NULL,
    QuestionNumber INT NOT NULL,
    QuestionText NVARCHAR(MAX) NOT NULL,
    Hint NVARCHAR(MAX) NULL,
    AllowShuffleQuestion BIT NOT NULL DEFAULT 0,
    AllowShuffleAnswers BIT NOT NULL DEFAULT 0,
    CONSTRAINT FK_Questions_Exams FOREIGN KEY (ExamId) 
        REFERENCES Exams(Id),
    CONSTRAINT FK_Questions_QuestionGroups FOREIGN KEY (QuestionGroupId) 
        REFERENCES QuestionGroups(Id) ON DELETE CASCADE
);

-- 4. BẢNG QUESTION OPTIONS (4 lựa chọn A, B, C, D)
CREATE TABLE QuestionOptions (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    QuestionId INT NOT NULL,
    OptionLabel VARCHAR(5) NOT NULL, -- 'A', 'B', 'C', 'D'
    OptionText NVARCHAR(MAX) NOT NULL,
    IsCorrect BIT NOT NULL DEFAULT 0,
    CONSTRAINT FK_QuestionOptions_Questions FOREIGN KEY (QuestionId) 
        REFERENCES Questions(Id) ON DELETE CASCADE
);

-- 5. BẢNG EXAM ATTEMPTS (Lượt làm bài của sinh viên)
CREATE TABLE ExamAttempts (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    ExamId INT NOT NULL,
    DeviceId NVARCHAR(100) NULL,
    StartTime DATETIME2 NOT NULL DEFAULT GETUTCDATE(),
    SubmitTime DATETIME2 NOT NULL DEFAULT GETUTCDATE(),
    TimeSpentSeconds INT NOT NULL DEFAULT 0,
    AnsweredCount INT NOT NULL DEFAULT 0,
    CorrectCount INT NOT NULL DEFAULT 0,
    WrongCount INT NOT NULL DEFAULT 0,
    SkippedCount INT NOT NULL DEFAULT 0,
    Score FLOAT NOT NULL DEFAULT 0,
    CONSTRAINT FK_ExamAttempts_Exams FOREIGN KEY (ExamId) 
        REFERENCES Exams(Id) ON DELETE CASCADE
);

-- 6. BẢNG EXAM ATTEMPT ANSWERS (Chi tiết từng câu nộp để thống kê)
CREATE TABLE ExamAttemptAnswers (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    ExamAttemptId INT NOT NULL,
    QuestionId INT NOT NULL,
    SelectedOptionId INT NULL,
    IsCorrect BIT NOT NULL DEFAULT 0,
    CONSTRAINT FK_ExamAttemptAnswers_ExamAttempts FOREIGN KEY (ExamAttemptId) 
        REFERENCES ExamAttempts(Id) ON DELETE CASCADE,
    CONSTRAINT FK_ExamAttemptAnswers_Questions FOREIGN KEY (QuestionId) 
        REFERENCES Questions(Id),
    CONSTRAINT FK_ExamAttemptAnswers_QuestionOptions FOREIGN KEY (SelectedOptionId) 
        REFERENCES QuestionOptions(Id)
);

-- INDEXES TỐI ƯU HIỆU NĂNG TRUY VẤN
CREATE INDEX IX_QuestionGroups_ExamId ON QuestionGroups(ExamId);
CREATE INDEX IX_Questions_ExamId_QuestionGroupId ON Questions(ExamId, QuestionGroupId);
CREATE INDEX IX_QuestionOptions_QuestionId ON QuestionOptions(QuestionId);
CREATE INDEX IX_ExamAttempts_ExamId ON ExamAttempts(ExamId);
CREATE INDEX IX_ExamAttemptAnswers_AttemptId_QuestionId ON ExamAttemptAnswers(ExamAttemptId, QuestionId);
